cmake_minimum_required(VERSION 3.21)
project(oma-audio-books VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Quick QuickControls2 Multimedia Sql Concurrent DBus Network)
qt_standard_project_setup()
qt_add_executable(oma-audio-books main.cpp app.cpp app.h scanner.cpp desktop.cpp desktop.h smoke.cpp)
qt_add_resources(oma-audio-books ui PREFIX "/" FILES Main.qml)
target_link_libraries(oma-audio-books PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Sql Qt6::Concurrent Qt6::DBus Qt6::Network)
target_compile_options(oma-audio-books PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
install(TARGETS oma-audio-books RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/oma-audio-books.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/oma-audio-books.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES README.md IMPLEMENTATION.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/oma-audio-books)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/oma-audio-books)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py $<TARGET_FILE:oma-audio-books>)
set_tests_properties(integration PROPERTIES TIMEOUT 120)

qt_add_executable(performance-check tests/performance.cpp app.cpp app.h scanner.cpp desktop.cpp desktop.h)
target_include_directories(performance-check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
qt_add_resources(performance-check ui PREFIX "/" FILES Main.qml)
target_link_libraries(performance-check PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Sql Qt6::Concurrent Qt6::DBus Qt6::Network)
target_compile_options(performance-check PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME performance COMMAND performance-check)
set_tests_properties(performance PROPERTIES TIMEOUT 120 ENVIRONMENT "QT_FATAL_WARNINGS=1")

add_executable(scanner-check tests/scanner.cpp scanner.cpp)
target_include_directories(scanner-check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(scanner-check PRIVATE Qt6::Core)
target_compile_options(scanner-check PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME scanner COMMAND scanner-check)
set_tests_properties(scanner PROPERTIES TIMEOUT 120)
